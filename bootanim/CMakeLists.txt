# x6100-bootanim: the boot logo composes itself while the radio boots (see bootanim.c)
find_package(PNG REQUIRED)
add_executable(x6100-bootanim bootanim.c)
if(BOARD_GSOC)
    # GSOC: 1024×600 landscape fb, centred 800×480 logo; no X6100 CW/GPIO sound path
    # (HW_BACKEND=g90 has no aether_x6100_control target).
    target_link_libraries(x6100-bootanim PRIVATE PNG::PNG pthread m)
else()
    if(HW_BACKEND STREQUAL "x6200")
        set(BOOT_CONTROL aether_x6200_control)
    else()
        set(BOOT_CONTROL aether_x6100_control)
    endif()
    target_link_libraries(x6100-bootanim PRIVATE PNG::PNG ${BOOT_CONTROL} gpiod asound sqlite3 pthread m)
endif()
install(TARGETS x6100-bootanim DESTINATION sbin)
install(FILES ${CMAKE_SOURCE_DIR}/boot/logo.png DESTINATION share/x6100 RENAME bootlogo.png)
